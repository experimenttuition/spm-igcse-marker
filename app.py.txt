import streamlit as st
from google import genai
from google.genai import types
import pdfplumber
import docx
import json
from fpdf import FPDF

# Initialize Gemini Client using Streamlit Secret
client = genai.Client(api_key=st.secrets["GEMINI_API_KEY"])

def extract_text(uploaded_file):
    """Extract text from uploaded PDF, DOCX, or TXT file."""
    if uploaded_file.name.endswith('.pdf'):
        with pdfplumber.open(uploaded_file) as pdf:
            return "\n".join([page.extract_text() for page in pdf.pages if page.extract_text()])
    elif uploaded_file.name.endswith('.docx'):
        doc = docx.Document(uploaded_file)
        return "\n".join([p.text for p in doc.paragraphs])
    else:
        return str(uploaded_file.read(), "utf-8")

def create_pdf(student_name, syllabus, word_count, prompt_text, data):
    """Generate downloadable PDF summary report using FPDF."""
    pdf = FPDF()
    pdf.add_page()
    
    # Title Header
    pdf.set_font("Arial", 'B', 16)
    pdf.set_text_color(30, 58, 138)
    pdf.cell(0, 10, "Writing Evaluation Report", ln=True)
    
    # Metadata
    pdf.set_font("Arial", '', 10)
    pdf.set_text_color(50, 50, 50)
    pdf.cell(0, 6, f"Student: {student_name} | Syllabus: {syllabus} | Words: {word_count}", ln=True)
    pdf.ln(4)
    
    # Score Box
    pdf.set_font("Arial", 'B', 14)
    pdf.set_text_color(29, 78, 216)
    pdf.cell(0, 8, f"Total Score: {data.get('overall_score', 0)} / {data.get('max_score', 0)}", ln=True)
    pdf.ln(4)
    
    # Criteria Breakdown
    pdf.set_font("Arial", 'B', 11)
    pdf.set_text_color(0, 0, 0)
    pdf.cell(0, 6, "Criteria Breakdown:", ln=True)
    
    pdf.set_font("Arial", '', 9)
    for item in data.get('breakdown', []):
        crit_line = f"- {item.get('criterion')}: {item.get('score')}/{item.get('max')} | {item.get('feedback')}"
        pdf.multi_cell(0, 5, crit_line.encode('latin-1', 'replace').decode('latin-1'))
    
    # Strengths Section
    pdf.ln(4)
    pdf.set_font("Arial", 'B', 11)
    pdf.cell(0, 6, "Key Strengths:", ln=True)
    pdf.set_font("Arial", '', 9)
    for s in data.get('strengths', []):
        pdf.multi_cell(0, 5, f"* {s}".encode('latin-1', 'replace').decode('latin-1'))

    # Improvements Section
    pdf.ln(4)
    pdf.set_font("Arial", 'B', 11)
    pdf.cell(0, 6, "Areas for Improvement:", ln=True)
    pdf.set_font("Arial", '', 9)
    for i in data.get('improvements', []):
        pdf.multi_cell(0, 5, f"* {i}".encode('latin-1', 'replace').decode('latin-1'))

    return bytes(pdf.output())

# Streamlit UI Configuration
st.set_page_config(page_title="SPM & IGCSE Essay Marker", layout="wide")
st.title("📝 Automated Writing Marker")

# Sidebar Controls
syllabus = st.sidebar.selectbox("Select Syllabus", ["SPM 1119", "IGCSE 0500"])
task_prompt = st.sidebar.text_area("Assignment Prompt (Optional)")

# File Upload Interface
uploaded_file = st.file_uploader("Upload Student Essay (.pdf, .docx, .txt)", type=["pdf", "docx", "txt"])

if uploaded_file and st.button("Mark Essay"):
    essay_text = extract_text(uploaded_file)
    word_count = len(essay_text.split())
    
    system_instruction = f"""
    You are an official examiner for {syllabus}. Evaluate the essay according to official rubric standards.
    Return JSON ONLY with this schema:
    {{
      "overall_score": 0,
      "max_score": 0,
      "breakdown": [
        {{"criterion": "Criterion Name", "score": 0, "max": 0, "feedback": "Detailed notes referencing text."}}
      ],
      "strengths": ["Strength 1"],
      "improvements": ["Improvement 1"]
    }}
    """
    
    with st.spinner("Grading essay against official criteria..."):
        try:
            response = client.models.generate_content(
                model="gemini-2.5-flash",
                contents=f"ASSIGNMENT PROMPT: {task_prompt}\n\nSTUDENT ESSAY:\n{essay_text}",
                config=types.GenerateContentConfig(
                    system_instruction=system_instruction,
                    response_mime_type="application/json"
                )
            )
            
            data = json.loads(response.text)
            
            # Display Evaluation Results
            st.header(f"Total Score: {data['overall_score']} / {data['max_score']}")
            
            st.subheader("Criteria Breakdown")
            for item in data['breakdown']:
                st.markdown(f"**{item['criterion']} ({item['score']}/{item['max']})**: {item['feedback']}")
                
            st.subheader("Key Strengths")
            for s in data['strengths']:
                st.write(f"- {s}")
                
            st.subheader("Areas for Improvement")
            for i in data['improvements']:
                st.write(f"- {i}")
                
            # PDF Download Button
            pdf_bytes = create_pdf(uploaded_file.name.split('.')[0], syllabus, word_count, task_prompt, data)
            st.download_button(
                label="📥 Download PDF Marking Sheet",
                data=pdf_bytes,
                file_name=f"Marked_{uploaded_file.name.split('.')[0]}.pdf",
                mime="application/pdf"
            )

        except Exception as e:
            st.error(f"Error evaluating essay: {e}")
